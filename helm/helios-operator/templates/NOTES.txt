1. Get the application URL by running these commands:
{{- if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Values.namespace.name }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "helios-operator.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Values.namespace.name }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Values.namespace.name }} svc -w {{ include "helios-operator.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Values.namespace.name }} {{ include "helios-operator.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Values.namespace.name }} -l "app.kubernetes.io/name={{ include "helios-operator.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Values.namespace.name }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Values.namespace.name }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Verify the HeliosApp CRD is installed:
   kubectl get crd heliosapps.platform.helios.io

3. Create a sample HeliosApp:
   cat <<EOF | kubectl apply -f -
   apiVersion: platform.helios.io/v1
   kind: HeliosApp
   metadata:
     name: sample-app
     namespace: default
   spec:
     gitRepo: "https://github.com/your-org/your-app.git"
     gitBranch: "main"
     gitopsRepo: "https://github.com/your-org/your-gitops.git"
     gitopsPath: "sample-app"
     imageRepo: "docker.io/your-org/sample-app"
     port: 80
     replicas: 1
     serviceAccount: "pipeline-sa"
     webhookSecret: "github-webhook-secret"
     pvcName: "sample-app-pvc"
   EOF

4. Check the operator logs:
   kubectl logs -n {{ .Values.namespace.name }} deployment/{{ include "helios-operator.fullname" . }}

{{- if .Values.metrics.enabled }}
5. Access metrics endpoint:
   kubectl port-forward -n {{ .Values.namespace.name }} svc/{{ include "helios-operator.fullname" . }}-metrics 8443:8443
   # Then visit: https://localhost:8443/metrics
{{- end }}

{{- if .Values.webhook.enabled }}
6. Webhook is enabled and will validate HeliosApp resources.
{{- end }}
